import { test, expect } from '@playwright/test';
import sharp from 'sharp';
import { BASE } from '../../tests/e2e/support/artifact';
import { promptDetail } from '../../tests/e2e/support/content';
for (const width of [390, 1440]) {
  for (const [id, language, loop] of [['p_236','en',null],['p_237','ja',null],['p_306','ja',0],['p_307','tr',1]] as const) {
    test(`${id} language ${language}, media, clipboard and export at ${width}px`, async ({page,context}) => {
      await page.setViewportSize({width,height:1000});
      await context.grantPermissions(['clipboard-read','clipboard-write']);
      for (const locale of ['en','zh-Hans']) {
        const detail=promptDetail(locale,id);
        await page.goto(`${BASE}${detail.path}`);
        const prompt=page.locator('[id$="-prompt"]').first();
        await expect(prompt).toHaveText(detail.prompt.text);
        await expect(prompt).toHaveAttribute('lang',language);
        await page.locator('[data-copy-state]:visible').first().click();
        expect(await page.evaluate(()=>navigator.clipboard.readText())).toBe(detail.prompt.text);
        const alternate=await page.locator('link[rel="alternate"][type="text/markdown"]').getAttribute('href');
        const response=await page.request.get(new URL(alternate!).pathname);
        expect(response.ok()).toBe(true);
        expect(/^(`{3,})text\n([\s\S]*?)\n\1$/m.exec(await response.text())?.[2]).toBe(detail.prompt.text);
        const hero=page.locator('[data-stage-picture] img').first();
        await expect.poll(()=>hero.evaluate((img:HTMLImageElement)=>img.complete&&img.naturalWidth>0)).toBe(true);
        const media=await page.request.get(await hero.evaluate((img:HTMLImageElement)=>img.currentSrc));
        expect(media.ok()).toBe(true);
        const metadata=await sharp(await media.body(),{animated:true}).metadata();
        if(loop!==null){expect(metadata.pages).toBeGreaterThanOrEqual(8);expect(metadata.loop).toBe(loop);expect(metadata.delay!.reduce((a,b)=>a+b,0)).toBe(2000);}
        else expect(metadata.pages??1).toBe(1);
        expect(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth)).toBe(true);
      }
    });
  }
}
